const cityInput = document.getElementById('cityInput'); const weatherBtn = document.getElementById('weatherBtn'); const weatherResult = document.getElementById('weatherResult'); let isLoading = false; const HOST = "me4ewvc2d7.re.qweatherapi.com"; const DEV_ID = "Q8A1B68E16"; // 这里必须填入你Web API的KEY！！ const KEY = "在此粘贴你的Web API凭据KEY"; weatherBtn.addEventListener('click', async ()=>{ const city = cityInput.value.trim(); if(!city){ weatherResult.innerText = "请输入城市名称"; return; } if(isLoading) return; isLoading = true; weatherBtn.disabled = true; weatherResult.innerText = "正在查询城市ID..."; try{ const geoUrl = `https://${HOST}/geo/v2/city/lookup?location=${encodeURIComponent(city)}&key=${KEY}`; const geoRes = await fetch(geoUrl); const geoJson = await geoRes.json(); console.log("返回的数据：", geoJson); // 控制台打印返回内容 if(!geoJson.code){ weatherResult.innerText = `返回数据异常，请查看控制台`; return; } if(geoJson.code !== "200"){ weatherResult.innerText = `城市查询失败：${geoJson.code}`; return; } const locationId = geoJson.location[0].id; weatherResult.innerText = "正在获取实时天气..."; const weatherUrl = `https://${HOST}/v7/weather/now?location=${locationId}&key=${KEY}`; const weatherRes = await fetch(weatherUrl); const weatherJson = await weatherRes.json(); console.log("天气数据：", weatherJson); if(weatherJson.code !== "200"){ weatherResult.innerText = `天气查询失败：${weatherJson.code}`; return; } const now = weatherJson.now; const cityName = geoJson.location[0].name; const outputText = `城市：${cityName} 天气：${now.text} 温度：${now.temp}℃ 体感温度：${now.feelsLike}℃ 风向：${now.windDir} 风力：${now.windScale}级 湿度：${now.humidity}%`; weatherResult.innerText = outputText; }catch(err){ console.error(err); weatherResult.innerText = "请求网络错误"; }finally{ isLoading = false; weatherBtn.disabled = false; } }) cityInput.addEventListener("keydown", e=>{ if(e.key === "Enter") weatherBtn.click(); })
